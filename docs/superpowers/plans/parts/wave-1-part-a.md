- [ ] **Step 14: Write scripts/check-tokens.mjs**

Ported from the Flow Kit app's `scripts/check-tokens.mjs`: same three patterns (raw hex, raw `oklch()`, Tailwind palette utility classes) and the same exit behavior (exit 1 on any violation, exit 0 otherwise), with the glob changed to `components/**/*.tsx` and the exclusion changed to `components/ui/**` plus `.test.` files.

```js
import { globSync, readFileSync } from "node:fs";

const FILES = globSync("components/**/*.tsx", {
  exclude: (f) => f.includes(".test.") || f.startsWith("components/ui/"),
});

// One entry per token-contract rule (design spec, Visual language section). Known limitation:
// issue refs like "#1234" in comments can false-positive as hex; use GH-1234 in component
// sources instead.
const PATTERNS = [
  { re: /#[0-9a-fA-F]{3,8}\b/g, why: "raw hex color" },
  { re: /\boklch\s*\(/gi, why: "raw oklch()" },
  {
    re: /\b(?:bg|text|border|ring|fill|stroke|from|via|to|outline|decoration|divide|accent|caret|shadow)-(?:slate|gray|zinc|neutral|stone|red|orange|amber|yellow|lime|green|emerald|teal|cyan|sky|blue|indigo|violet|purple|fuchsia|pink|rose)-\d{2,3}\b/g,
    why: "tailwind palette class",
  },
];

if (FILES.length === 0) {
  console.warn(
    "check:tokens: WARNING: no .tsx files found under components/. Gate has no coverage yet.",
  );
}

let violations = 0;
for (const file of FILES) {
  let source;
  try {
    source = readFileSync(file, "utf8");
  } catch (e) {
    violations++;
    console.error(`${file}: could not read (${e.code ?? e.message})`);
    continue;
  }
  source.split("\n").forEach((line, i) => {
    for (const { re, why } of PATTERNS) {
      if (re.test(line)) {
        violations++;
        console.error(`${file}:${i + 1}: ${why}: ${line.trim()}`);
      }
      re.lastIndex = 0;
    }
  });
}

if (violations) {
  console.error(`\ncheck:tokens: ${violations} violation(s). Use shadcn CSS variables.`);
  process.exit(1);
}
console.log(`check:tokens: ${FILES.length} file(s) clean.`);
```

```ts
export type HandleDir = "in" | "out";

const PORT_RE = /^[a-z0-9-]+$/;

/** Handle id codec: `{nodeId}:{dataType}:{in|out}`, or with a port `{nodeId}:{dataType}:{in|out}@{port}`. */
export function handleId(nodeId: string, dataType: string, dir: HandleDir, port?: string): string {
  const base = `${nodeId}:${dataType}:${dir}`;
  if (port === undefined) return base;
  if (!PORT_RE.test(port)) {
    if (process.env.NODE_ENV !== "production") {
      console.warn(`handleId: invalid port "${port}" - use lowercase letters, digits, hyphens; omitting suffix`);
    }
    return base;
  }
  return `${base}@${port}`;
}

export function parseHandleId(id: string | null | undefined) {
  if (!id) return null;
  const parts = id.split(":");
  if (parts.length < 3) return null;
  const dirSegment = parts.pop()!;
  const dataType = parts.pop()!;
  const nodeId = parts.join(":");
  const dirMatch = dirSegment.match(/^(in|out)(?:@(.+))?$/);
  if (!nodeId || !dataType || !dirMatch) return null;
  const dir = dirMatch[1] as HandleDir;
  const port = dirMatch[2];
  if (port !== undefined) {
    if (!PORT_RE.test(port)) return null;
    return { nodeId, dataType, dir, port } as { nodeId: string; dataType: string; dir: HandleDir; port: string };
  }
  return { nodeId, dataType, dir } as { nodeId: string; dataType: string; dir: HandleDir };
}
```

```tsx
"use client";
import { Handle, Position, useNodeId, type Connection, type Edge, type HandleProps } from "@xyflow/react";
import { cn } from "@/lib/utils";
import { getHandleType, handleId, isValidFlowConnection } from "./flow-types";

export interface TypedHandleProps extends Omit<
  HandleProps,
  "type" | "position" | "id" | "isValidConnection"
> {
  /**
   * The React Flow node id that owns this handle.
   *
   * Optional — when rendered inside a React Flow node context the id is read
   * automatically via `useNodeId()`. Only pass this prop outside that context
   * (e.g. unit tests, Storybook, or standalone demos).
   */
  nodeId?: string;
  dataType: string;
  type: "source" | "target";
  position?: Position;
  /** vertical offset when stacking multiple ports on one side */
  top?: number;
  className?: string;
  /** Optional port name, appended to the handle id as `@{port}`. */
  port?: string;
  /** Extra validator ANDed with the built-in isValidFlowConnection. */
  validate?: (c: Connection | Edge) => boolean;
}

/**
 * A typed React Flow handle that encodes data-type semantics into the handle id
 * and enforces same-type, out→in connection validation.
 *
 * ### Handle id codec
 * Rendered handle ids follow the pattern `{nodeId}:{dataType}:{in|out}`, or
 * `{nodeId}:{dataType}:{in|out}@{port}` when a `port` prop is given.
 * This codec is consumed by typed-edge color derivation and connection
 * validation — do not bypass it by setting the `id` prop directly.
 *
 * ### Connection validation
 * Same-type, out to in connection validation is always applied and cannot be
 * bypassed. Pass `validate` to add extra rules; it is ANDed with the built-in
 * `isValidFlowConnection`, so a `validate` that returns `true` can never
 * approve a connection the built-in check rejects.
 *
 * ### Drag-state highlighting
 * `data-flow-compatible="true"` on the rendered handle is the extension hook
 * for drag-state highlighting (scale-up). Hosts can set it via
 * `useConnection()` from `@xyflow/react`. The companion "incompatible → dim"
 * treatment is deferred and recorded in the wave spec deviations.
 *
 * ### Custom type colors
 * Overriding `style.background` breaks type-color semantics. Register a custom
 * type via `registerHandleType` from `./flow-types` instead.
 */
export function TypedHandle({
  nodeId,
  dataType,
  type,
  position,
  top,
  className,
  style,
  port,
  validate,
  ...rest
}: TypedHandleProps) {
  const contextNodeId = useNodeId();
  const nid = nodeId ?? contextNodeId ?? "";
  if (process.env.NODE_ENV !== "production" && nid === "") {
    console.warn("TypedHandle: no nodeId available — handle id codec will be invalid");
  }
  const def = getHandleType(dataType);
  const dir = type === "source" ? "out" : "in";
  const isValidConnection = validate
    ? (c: Connection | Edge) => isValidFlowConnection(c) && validate(c)
    : isValidFlowConnection;
  return (
    <Handle
      id={handleId(nid, dataType, dir, port)}
      type={type}
      position={position ?? (type === "source" ? Position.Right : Position.Left)}
      isValidConnection={isValidConnection}
      aria-label={`${def?.label ?? dataType} ${dir === "in" ? "input" : "output"} port`}
      data-slot="typed-handle"
      data-flow-type={dataType}
      className={cn(
        "size-3.5 rounded-full border-2 border-background transition-transform",
        "data-[flow-compatible=true]:scale-125",
        className,
      )}
      style={{ background: `var(${def?.cssVar ?? "--flow-text"})`, ...(top != null && { top }), ...style }}
      {...rest}
    />
  );
}
```

```bash
#!/usr/bin/env bash
# scripts/flow-kit-diff.sh
# Compares each vendored components/flow-kit/ file against its source in flow-kit-worktree,
# ignoring the provenance header line this repo prepends to the vendored copy. Informational
# only: always exits 0, so it is safe to run in CI or locally without failing a build on
# upstream drift.
set -u

WORKTREE="${FLOW_KIT_WORKTREE:-/Users/nickv/ClaudeCode Projects/flow-kit-worktree}"
SRC_DIR="$WORKTREE/apps/docs/registry/super-ai/flow"
DST_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)/components/flow-kit"

FILES=(
  flow-types.ts
  flow-types.test.ts
  flow-tokens.css
  typed-handle.tsx
  typed-handle.test.tsx
  typed-edge.tsx
  typed-edge.test.tsx
)

drifted=0
for f in "${FILES[@]}"; do
  src="$SRC_DIR/$f"
  dst="$DST_DIR/$f"
  if [ ! -f "$src" ]; then
    echo "SKIP  $f (worktree source not found at $src)"
    continue
  fi
  if [ ! -f "$dst" ]; then
    echo "SKIP  $f (vendored file not found at $dst)"
    continue
  fi
  if diff -q "$src" <(tail -n +2 "$dst") >/dev/null; then
    echo "OK    $f"
  else
    drifted=$((drifted + 1))
    echo "DIFF  $f"
    diff -u "$src" <(tail -n +2 "$dst") | sed 's/^/      /'
  fi
done

echo ""
echo "flow-kit-diff: ${#FILES[@]} file(s) checked, $drifted with content drift beyond the header."
exit 0
```

```ts
      ctx.addIssue({ code: "custom", path: ["nodes", i, "kind"], message: `unknown kind "${node.kind}"` });

      }

      ctx.addIssue({
        code: "custom",
        path: ["nodes", i, "parentId"],
        message: `parentId "${node.parentId}" does not reference a "while" node`,
      });
    }

    if (!nodeIds.has(edge.source)) {
      ctx.addIssue({ code: "custom", path: ["edges", i, "source"], message: `source "${edge.source}" does not reference an existing node` });
    }
    if (!nodeIds.has(edge.target)) {
      ctx.addIssue({ code: "custom", path: ["edges", i, "target"], message: `target "${edge.target}" does not reference an existing node` });
    }
    const sourceParsed = parseHandleId(edge.sourceHandle);

        path: ["edges", i, "sourceHandle"],

      });

export type ParseResult = { ok: true; flow: Flow } | { ok: false; errors: string[] };

function formatPath(path: (string | number)[]): string {
  let out = "";
  for (const seg of path) {
    if (typeof seg === "number") out += `[${seg}]`;
    else out += out ? `.${seg}` : seg;
  }
  return out;
}

export function parseFlow(json: unknown): ParseResult {
  const migrated = migrateFlow(json);
  const result = flowSchema.safeParse(migrated);
  if (result.success) return { ok: true, flow: result.data as Flow };
  const errors = result.error.issues.map((issue) => {
    const path = formatPath(issue.path as (string | number)[]);
    return path ? `${path}: ${issue.message}` : issue.message;
  });
  return { ok: false, errors };
}

export function migrateFlow(json: unknown): unknown {
  const version = (json as { schemaVersion?: unknown } | null)?.schemaVersion;
  if (version === 1 || version === undefined) return json;
  throw new Error(`unsupported schemaVersion ${String(version)}`);
}

function randomSlug(length: number): string {
  const chars = "abcdefghijklmnopqrstuvwxyz0123456789";
  let out = "";
  for (let i = 0; i < length; i++) out += chars[Math.floor(Math.random() * chars.length)];
  return out;
}
```

```ts
// lib/flow-convert.ts
import type { Edge, Node } from "@xyflow/react";
import { parseHandleId } from "@/components/flow-kit/flow-types";
import type { AnyKindData, Flow, FlowEdge, FlowMeta, FlowNode, FlowViewport, NodeKind } from "./flow-schema";

export type BuilderNode = Node<AnyKindData, NodeKind>;
export type BuilderEdge = Edge<{ port?: string; streaming?: boolean }, "typed">;

export function toBuilderNode(n: FlowNode): BuilderNode {
  const node: BuilderNode = {
    id: n.id,
    type: n.kind,
    position: n.position,
    data: n.data,
  };
  if (n.parentId !== undefined) {
    node.parentId = n.parentId;
    node.extent = "parent";
  }
  if (n.kind === "note") node.zIndex = -1;
  if (n.width !== undefined) node.width = n.width;
  if (n.height !== undefined) node.height = n.height;
  return node;
}

export function fromBuilderNode(n: BuilderNode): FlowNode {
  const node: FlowNode = {
    id: n.id,
    kind: n.type as NodeKind,
    position: n.position,
    data: n.data,
  };
  if (n.parentId !== undefined) node.parentId = n.parentId;
  if (n.width !== undefined) node.width = n.width;
  if (n.height !== undefined) node.height = n.height;
  return node;
}

export function toBuilderEdge(e: FlowEdge): BuilderEdge {
  const port = parseHandleId(e.sourceHandle)?.port;
  return {
    id: e.id,
    type: "typed",
    source: e.source,
    target: e.target,
    sourceHandle: e.sourceHandle,
    targetHandle: e.targetHandle,
    ...(e.label !== undefined && { label: e.label }),
    data: { ...(port !== undefined && { port }) },
  };
}

export function fromBuilderEdge(e: BuilderEdge): FlowEdge {
  const edge: FlowEdge = {
    id: e.id,
    source: e.source,
    target: e.target,
    sourceHandle: e.sourceHandle ?? "",
    targetHandle: e.targetHandle ?? "",
  };
  if (typeof e.label === "string") edge.label = e.label;
  return edge;
}

export function sortParentsFirst(nodes: BuilderNode[]): BuilderNode[] {
  const byId = new Map(nodes.map((n) => [n.id, n]));
  const depth = (n: BuilderNode): number => {
    let d = 0;
    let current: BuilderNode | undefined = n;
    while (current?.parentId) {
      current = byId.get(current.parentId);
      d++;
    }
    return d;
  };
  return nodes
    .map((n, i) => ({ n, i, d: depth(n) }))
    .sort((a, b) => a.d - b.d || a.i - b.i)
    .map((x) => x.n);
}

export function flowToGraph(flow: Flow): { nodes: BuilderNode[]; edges: BuilderEdge[] } {
  return {
    nodes: sortParentsFirst(flow.nodes.map(toBuilderNode)),
    edges: flow.edges.map(toBuilderEdge),
  };
}

export function graphToFlow(
  meta: FlowMeta,
  nodes: BuilderNode[],
  edges: BuilderEdge[],
  viewport?: FlowViewport,
): Flow {
  return {
    schemaVersion: 1,
    ...meta,
    nodes: sortParentsFirst(nodes).map(fromBuilderNode),
    edges: edges.map(fromBuilderEdge),
    ...(viewport !== undefined && { viewport }),
  };
}

export function canonicalFlowJson(flow: Flow): string {
  return JSON.stringify(sortKeysDeep(flow), null, 2) + "\n";
}

function sortKeysDeep(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(sortKeysDeep);
  if (value !== null && typeof value === "object") {
    const sorted: Record<string, unknown> = {};
    for (const key of Object.keys(value as Record<string, unknown>).sort()) {
      sorted[key] = sortKeysDeep((value as Record<string, unknown>)[key]);
    }
    return sorted;
  }
  return value;
}
```

```ts
// lib/flows.server.ts
import { readdir, readFile } from "node:fs/promises";
import path from "node:path";
import { parseFlow, type Flow } from "./flow-schema";

export const FLOWS_DIR = path.join(process.cwd(), "flows");

export async function listFlows(dir: string = FLOWS_DIR): Promise<Flow[]> {
  const entries = await readdir(dir);
  const files = entries.filter((f) => f.endsWith(".json")).sort();
  const flows: Flow[] = [];
  for (const file of files) {
    const raw = await readFile(path.join(dir, file), "utf8");
    const result = parseFlow(JSON.parse(raw));
    if (!result.ok) {
      throw new Error(`${file}: ${result.errors.join("; ")}`);
    }
    flows.push(result.flow);
  }
  return flows;
}

export async function getFlow(slug: string, dir: string = FLOWS_DIR): Promise<Flow | null> {
  const file = path.join(dir, `${slug}.json`);
  let raw: string;
  try {
    raw = await readFile(file, "utf8");
  } catch {
    return null;
  }
  const result = parseFlow(JSON.parse(raw));
  if (!result.ok) {
    throw new Error(`${slug}.json: ${result.errors.join("; ")}`);
  }
  return result.flow;
}
```
